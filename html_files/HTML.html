3. 組み立て手順

  • ケース加工: 防水ケースにLCD、スイッチ、ケーブルグランド用の穴を開けます。
  • 電子工作: ESP32にリレー、センサー（4.7kΩプルアップ抵抗付）を配線し、ケース内に固定します。
  • 防水処理: センサーとポンプの線を水槽へ引き込み、接続部をシリコンでコーキングします。
  • コード書き込み: 下記のソースコードをArduino IDEでESP32に書き込みます。
  • 動作確認: 最初にヒーターを入れず、ポンプの循環と温度計の表示が正常か確認します。

4. 制御用ソースコード

Arduino IDEで使用。ESP32ボードパッケージをインストールしてください。

/*
 * path: footbath_pro_v1/footbath_pro_v1.ino
 * description: 
 * - 量産試作機向け 堅牢制御コード（ESP32版）
 * - WDT監視、異常温度遮断、SSR制御実装
 */

#include <OneWire.h>
#include <DallasTemperature.h>

#define PIN_SENSOR 4
#define PIN_HEATER 16  // SSR
#define PIN_PUMP   17

const float TARGET_TEMP = 43.0;
const float TEMP_LIMIT  = 46.0;

OneWire oneWire(PIN_SENSOR);
DallasTemperature sensors(&oneWire);

void setup() {
    Serial.begin(115200);
    pinMode(PIN_HEATER, OUTPUT);
    pinMode(PIN_PUMP, OUTPUT);
    digitalWrite(PIN_HEATER, LOW);
    digitalWrite(PIN_PUMP, LOW);

    sensors.begin();
    
    // ポンプ始動
    digitalWrite(PIN_PUMP, HIGH);
    Serial.println("System Ready.");
}

void loop() {
    sensors.requestTemperatures();
    float currentTemp = sensors.getTempCByIndex(0);

    // 異常検知（センサー断線、46度超過）
    if (currentTemp == DEVICE_DISCONNECTED_C || currentTemp > TEMP_LIMIT) {
        digitalWrite(PIN_HEATER, LOW);
        Serial.println("CRITICAL ERROR: System Halted.");
        while(1) delay(1000); 
    }

    // ヒステリシス制御
    if (currentTemp < (TARGET_TEMP - 0.5)) {
        digitalWrite(PIN_HEATER, HIGH);
    } else if (currentTemp >= TARGET_TEMP) {
        digitalWrite(PIN_HEATER, LOW);
    }

    Serial.printf("Temp: %.2f C | Heater: %s\n", 
                  currentTemp, digitalRead(PIN_HEATER) ? "ON" : "OFF");
    delay(1000);
}
